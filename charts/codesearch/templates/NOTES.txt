codesearch {{ .Chart.AppVersion }} is running as {{ include "codesearch.fullname" . }} in namespace {{ .Release.Namespace }}.

1. Read the API key:

  export CODESEARCH_API_KEY=$(kubectl get secret -n {{ .Release.Namespace }} {{ include "codesearch.apiKeySecretName" . }} -o jsonpath='{.data.{{ include "codesearch.apiKeySecretKey" . }}}' | base64 -d)

2. Reach the server:
{{- if .Values.ingress.enabled }}
{{- with (first .Values.ingress.hosts) }}

  export CODESEARCH_URL=http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}
{{- end }}
{{- else }}

  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ include "codesearch.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
  export CODESEARCH_URL=http://localhost:{{ .Values.service.port }}
{{- end }}

  In-cluster clients use http://{{ include "codesearch.fullname" . }}.{{ .Release.Namespace }}.svc.{{ .Values.clusterDomain }}:{{ .Values.service.port }}

3. Connect Claude Code:

  claude mcp add --transport http codesearch "$CODESEARCH_URL/mcp" --header "Authorization: Bearer $CODESEARCH_API_KEY"
